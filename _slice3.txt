46|def cmd_dev(args) -> int:
47|    """一键起开发环境。
48|
49|    做了四件事：可选构建前端 → 确保 Redis → 起服务 →（可选）开窗口。
50|    Ctrl+C 的时候把自己起的 Redis 一起收掉，不留孤儿进程。
51|    """
52|    import uvicorn
53|
54|    from minicode import dev
55|    from minicode.app_shell import run_app
56|    from minicode.service import create_app
57|
58|    root = _repo_root()
59|
60|    # 端口被占用时给一句人话提示，而不是让 uvicorn 抛一屏 WinError 10048
61|    from minicode.dev import port_open
62|
63|    if port_open(args.host, args.port):
64|        print(f"端口 {args.port} 已经被占用了。")
65|        print("  可能你之前的 minicode dev 还开着 —— 关掉那个窗口再试。")
66|        print(f"  或者换个端口：minicode dev --port {args.port + 1}")
67|        return 1
68|
69|    if args.build:
70|        dev.build_frontend(root)
71|
72|    redis_proc, url = dev.ensure_redis(root, enabled=not args.no_redis)
73|    if args.no_redis:
74|        # 显式清掉：否则外部设置的 MINICODE_REDIS_URL 会让服务照样连 Redis，
75|        # 和用户明确要求的内存模式不符
76|        os.environ.pop("MINICODE_REDIS_URL", None)
77|    dev.apply_env(url)
78|    if url:
79|        state = "（已在运行，直接复用）" if redis_proc is None else "（刚启动）"
80|        print(f"Redis: {url} {state}")
81|    else:
82|        print("Redis: 没启用（用内存存储 —— 服务重启后历史会丢）")
83|
84|    app = create_app(default_workspace=args.workspace, token=args.token)
85|    print(f"服务: http://{args.host}:{args.port}/")
86|    if args.token:
87|        print("访问令牌已启用：界面右上角（令牌一栏）填同一个值")
88|    if args.open:
89|        print("按 Ctrl+C 结束（会一起收掉 Redis）")
90|    else:
91|        print("浏览器打开上面的地址；按 Ctrl+C 结束（会一起收掉 Redis）")
92|
93|    try:
94|        if args.open:
95|            return run_app(
96|                app,
97|                host=args.host,
98|                port=args.port,
99|                workspace=args.workspace,
100|                mode=args.mode,
101|            )
102|
103|        if args.reload:
104|            # reload 需要一个 import 字符串，uvicorn 会在子进程里重建 app。
105|            # 注意 factory 拿不到 --workspace，所以这里显式把它写进环境变量，
106|            # 让 create_app 的默认工作区跟着走（否则 --reload 会静默忽略 --workspace）。
107|            if args.workspace:
108|                os.environ["MINICODE_WORKSPACE"] = str(Path(args.workspace).resolve())
109|            uvicorn.run(
110|                "minicode.service:create_app",
111|                factory=True,
112|                host=args.host,
113|                port=args.port,
114|                reload=True,
115|                reload_dirs=[str(root / "src")],
116|            )
117|        else:
118|            uvicorn.run(app, host=args.host, port=args.port, log_level="info")
119|        return 0
120|    except KeyboardInterrupt:
121|        return 0
122|    finally:
123|        dev.stop(redis_proc)
